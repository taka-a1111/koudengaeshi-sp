/* 香典返しe-shop ／ 商品詳細
   - demoに合わせて価格の円マーク（U+00A5 / U+FFE5）を除去し「2,970円」表記に統一
   - ※ ShopServe管理画面は SJIS で円記号(U+00A5)とバックスラッシュ(U+005C)を同コードで扱うため、
       JS内では必ずUnicodeエスケープ（\u00A5, \uFFE5）で記述すること。生の半角円記号(U+00A5)は使わない。
   - フォーム文言、個数、通信欄案内を整形
   - カートボタンは ShopServe 標準（黒）をそのまま使用、商品名の【送料無料】も保持 */
document.addEventListener('DOMContentLoaded', function () {
  var piCart = document.querySelector('#pi_cart');
  if (!piCart) return;
  /* 価格表記を「(円記号)2,970」→「2,970円」へ。ShopServeの遅延描画＆再描画の両方に対応。
     (税込)等の非数値テキストや標準カートボタンには一切触れない。冪等。 */
  (function(){
    var YEN = /[\u00A5\uFFE5]/;          /* 円記号（半角・全角どちらも） */
    var YEN_G = /[\u00A5\uFFE5]\s*/g;
    var YEN_NUM_G = /[\u00A5\uFFE5]\s*([0-9][0-9,]*)(\u5186)?/g; /* \u5186 = 円 */
    function convertEl(el){
      if(!el) return;
      var raw = el.textContent;
      /* 円記号が無く既に「円」が付いていればスキップ */
      if(!YEN.test(raw) && /\u5186/.test(raw)) return;
      var walker = document.createTreeWalker(el, NodeFilter.SHOW_TEXT, null);
      var nodes = [];
      while(walker.nextNode()) nodes.push(walker.currentNode);
      nodes.forEach(function(n){
        var v = n.textContent;
        if(YEN.test(v)){
          if(/[\u00A5\uFFE5]\s*[0-9]/.test(v)){
            v = v.replace(YEN_NUM_G, function(_, num, yen){ return num + (yen ? yen : '\u5186'); });
          } else {
            v = v.replace(YEN_G, '');
          }
          if(v !== n.textContent) n.textContent = v;
        }
      });
      /* 数字のみで「円」が無ければ末尾に「円」を付与（円記号が別ノードだった場合の保険） */
      var full = el.textContent;
      if(/[0-9]/.test(full) && !/\u5186/.test(full) && !YEN.test(full)){
        var leaf = nodes.filter(function(n){ return /[0-9]/.test(n.textContent); }).pop();

      }
